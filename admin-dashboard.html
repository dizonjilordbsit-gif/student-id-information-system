-- Admin can view student records CREATE POLICY "Admins can view student records" ON public.students FOR SELECT TO authenticated USING ( EXISTS ( SELECT 1 FROM public.admin_users WHERE admin_users.user_id = auth.uid() ) ); -- Admin can view submitted documents CREATE POLICY "Admins can view documents" ON public.documents FOR SELECT TO authenticated USING ( EXISTS ( SELECT 1 FROM public.admin_users WHERE admin_users.user_id = auth.uid() ) ); -- Admin can update document status CREATE POLICY "Admins can update documents" ON public.documents FOR UPDATE TO authenticated USING ( EXISTS ( SELECT 1 FROM public.admin_users WHERE admin_users.user_id = auth.uid() ) ) WITH CHECK ( EXISTS ( SELECT 1 FROM public.admin_users WHERE admin_users.user_id = auth.uid() ) ); GRANT SELECT ON public.students TO authenticated; GRANT SELECT, UPDATE ON public.documents TO authenticated;
